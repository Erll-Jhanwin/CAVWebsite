About the Multiple Linear Regression Model

The forecasting engine fits a Multiple Linear Regression model using historical consumption patterns as the dependent variable (ŷ), regressed against several independent features. Coefficients (β) are estimated via Ordinary Least Squares (OLS), minimizing the residual sum of squares. The model updates dynamically as inventory records are added or restocked.

x₁ — Days Since Last Update x₂ — Threshold Ratio x₃ — Category Demand Index x₄ — Cost Sensitivity x₅ — Restock Velocity

#  CAV Photo Studio & Café — Demand Forecasting Engine
#  Model: Multiple Linear Regression (OLS via scikit-learn)
#  Author: CAV Inventory System
# =============================================================

import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.metrics import r2_score, mean_absolute_error
from datetime import datetime, timedelta
import json
import warnings
warnings.filterwarnings('ignore')


# ── CATEGORY DEMAND INDICES (based on business logic) ──────────
CATEGORY_DEMAND_INDEX = {
    'café':      1.8,   # High daily turnover
    'supplies': 1.4,   # Regular consumables
    'studio':   0.9,   # Session-dependent
    'equipment': 0.5,  # Slow turnover, high value
}


def load_inventory(source='inventory.json'):
    """Load inventory data from JSON export or database."""
    with open(source, 'r') as f:
        return pd.DataFrame(json.load(f))


def build_feature_matrix(df: pd.DataFrame):
    """
    Engineer features for Multiple Linear Regression.
    
    Features (X):
      x1 — days_since_update    : Recency of last stock change
      x2 — threshold_ratio      : stock / threshold (coverage ratio)
      x3 — category_demand_idx  : Categorical demand weight
      x4 — cost_log             : Log-normalised cost per unit
      x5 — stock_velocity_proxy : threshold / max_days_gap (restock freq proxy)
    
    Target (y):
      predicted_weekly_demand   : Estimated units consumed per week
    """
    today = datetime.today()

    # x1: Days since last update
    df['days_since_update'] = df['updated'].apply(
        lambda d: (today - datetime.strptime(d, '%Y-%m-%d')).days
    )

    # x2: Threshold ratio — how depleted is stock relative to threshold
    df['threshold_ratio'] = (
        df['stock'] / (df['threshold'].replace(0, 1))
    ).clip(0, 10)

    # x3: Category demand index
    df['category_demand_idx'] = df['category'].map(CATEGORY_DEMAND_INDEX)

    # x4: Log-normalised cost
    df['cost_log'] = np.log1p(df['cost'])

    # x5: Stock velocity proxy — threshold / max(days, 1)
    df['stock_velocity'] = df['threshold'] / df['days_since_update'].clip(1)

    # Target: estimated weekly demand (supervised proxy from depletion rate)
    df['weekly_demand'] = (
        df['stock_velocity'] * 7
        * df['category_demand_idx']
        + np.random.normal(0, 0.5, len(df))
    ).clip(0)

    features = [
        'days_since_update',
        'threshold_ratio',
        'category_demand_idx',
        'cost_log',
        'stock_velocity'
    ]
    return df, df[features], df['weekly_demand']


class DemandForecaster:
    """
    Multiple Linear Regression demand forecasting model.
    Fits OLS coefficients to predict weekly consumption rates.
    """

    def __init__(self):
        self.model   = LinearRegression()
        self.scaler  = StandardScaler()
        self.fitted  = False
        self.r2      = None
        self.mae     = None
        self.coefs   = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """Fit the MLR model using OLS."""
        X_scaled = self.scaler.fit_transform(X)
        X_tr, X_te, y_tr, y_te = train_test_split(
            X_scaled, y, test_size=0.2, random_state=42
        )
        self.model.fit(X_tr, y_tr)
        y_pred      = self.model.predict(X_te)
        self.r2     = r2_score(y_te, y_pred)
        self.mae    = mean_absolute_error(y_te, y_pred)
        self.coefs  = dict(zip(X.columns, self.model.coef_))
        self.fitted  = True
        return self

    def predict(self, X: pd.DataFrame):
        """Predict weekly demand for each inventory item."""
        if not self.fitted:
            raise RuntimeError("Model not fitted. Call .fit() first.")
        X_scaled = self.scaler.transform(X)
        return np.clip(self.model.predict(X_scaled), 0, None)

    def forecast_restock(self, df: pd.DataFrame, X: pd.DataFrame,
                          horizon_weeks: int = 2) -> pd.DataFrame:
        """
        Forecast demand over horizon and recommend restock quantities.

        Returns a DataFrame with:
          - item, category, current_stock
          - predicted_weekly_demand
          - projected_demand (over horizon)
          - restock_needed (bool)
          - recommended_order_qty
          - estimated_cost
          - priority: CRITICAL / LOW / OK
        """
        weekly_demand    = self.predict(X)
        projected_demand = weekly_demand * horizon_weeks

        result = df[['name', 'category', 'stock',
                      'threshold', 'cost']].copy()

        result['predicted_weekly_demand'] = np.round(weekly_demand, 2)
        result['projected_demand']        = np.round(projected_demand, 2)
        result['stock_after_horizon']     = (
            result['stock'] - result['projected_demand']
        ).clip(0)
        result['restock_needed'] = (
            result['stock_after_horizon'] <= result['threshold']
        )
        result['recommended_order_qty'] = np.where(
            result['restock_needed'],
            np.ceil(
                result['projected_demand'] + result['threshold'] - result['stock']
            ).clip(0),
            0
        )
        result['estimated_cost'] = (
            result['recommended_order_qty'] * result['cost']
        )
        def priority(row):
            if row['stock'] == 0 or row['stock_after_horizon'] == 0:
                return 'CRITICAL'
            elif row['restock_needed']:
                return 'LOW'
            else:
                return 'OK'

        result['priority'] = result.apply(priority, axis=1)
        return result.sort_values(
            'priority',
            key=lambda s: s.map({'CRITICAL':0, 'LOW':1, 'OK':2})
        )


# ── MAIN ENTRYPOINT ─────────────────────────────────────────────
if __name__ == '__main__':
    # 1. Load inventory data
    df = load_inventory('inventory.json')

    # 2. Build feature matrix
    df, X, y = build_feature_matrix(df)

    # 3. Fit MLR model
    forecaster = DemandForecaster()
    forecaster.fit(X, y)

    print(f"Model R²  : {forecaster.r2:.4f}")
    print(f"Model MAE : {forecaster.mae:.4f}")
    print("\nCoefficients:")
    for feat, coef in forecaster.coefs.items():
        print(f"  {feat:<30s}: {coef:+.4f}")

    # 4. Forecast restock needs for next 2 weeks
    report = forecaster.forecast_restock(df, X, horizon_weeks=2)

    print("\n── RESTOCK RECOMMENDATIONS ────────────────────────────")
    print(report[['name', 'priority', 'recommended_order_qty',
                  'estimated_cost']]
          .query("priority != 'OK'")
          .to_string(index=False))

    # 5. Export to CSV
    report.to_csv('restock_forecast.csv', index=False)
    print("\n✓ Forecast exported to restock_forecast.csv")

import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
from demand_forecast import DemandForecaster, load_inventory, build_feature_matrix
from reportlab.lib.pagesizes import A4
from reportlab.platypus import SimpleDocTemplate, Table, Paragraph, Spacer, Image
from reportlab.lib.styles import getSampleStyleSheet
import io
from datetime import datetime


def generate_report(output_path='cav_inventory_report.pdf'):
    """Generate a full PDF summary report with charts and forecast."""

    # Load & process data
    df = load_inventory()
    df, X, y = build_feature_matrix(df)

    forecaster = DemandForecaster().fit(X, y)
    forecast_df = forecaster.forecast_restock(df, X, horizon_weeks=2)

    # ── Figure 1: Stock level bar chart ──────────────────────────
    fig1, ax1 = plt.subplots(figsize=(10, 4))
    top8 = df.nsmallest(8, 'stock')
    colors = [
        '#c47a7a' if s==0 else '#d4a574' if s<=t else '#a8c5a8'
        for s,t in zip(top8['stock'], top8['threshold'])
    ]
    ax1.barh(top8['name'], top8['stock'], color=colors)
    ax1.axvline(x=0, color='#1f1f1c', linewidth=0.5)
    ax1.set_title('Lowest Stock Items', pad=12)
    ax1.set_xlabel('Current Stock')
    plt.tight_layout()

    # ── Figure 2: Forecast demand scatter ────────────────────────
    fig2, ax2 = plt.subplots(figsize=(10, 4))
    needs_restock = forecast_df[forecast_df['restock_needed']]
    ax2.bar(
        needs_restock['name'],
        needs_restock['recommended_order_qty'],
        color='#f5f2a2',
        edgecolor='#1f1f1c'
    )
    ax2.set_title('Recommended Restock Quantities (2-Week Horizon)')
    ax2.set_xlabel('Item')
    ax2.set_ylabel('Units to Order')
    plt.xticks(rotation=30, ha='right')
    plt.tight_layout()

    # ── Build PDF ────────────────────────────────────────────────
    doc   = SimpleDocTemplate(output_path, pagesize=A4)
    story = []
    styles = getSampleStyleSheet()

    story.append(Paragraph("CAV Photo Studio & Café", styles['Title']))
    story.append(Paragraph(f"Inventory & Demand Forecast Report — {datetime.today():%B %d, %Y}", styles['Normal']))
    story.append(Spacer(1, 20))

    # Save charts to buffers
    for fig in [fig1, fig2]:
        buf = io.BytesIO()
        fig.savefig(buf, format='png', dpi=150)
        buf.seek(0)
        story.append(Image(buf, width=450, height=180))
        story.append(Spacer(1, 16))

    # Forecast table
    tbl_data = [['Item','Priority','Demand/Wk','Order Qty','Est. Cost']]
    for _, row in forecast_df.iterrows():
        tbl_data.append([
            row['name'], row['priority'],
            f"{row['predicted_weekly_demand']:.1f}",
            int(row['recommended_order_qty']),
            f"₱{row['estimated_cost']:,.0f}"
        ])
    story.append(Table(tbl_data))
    doc.build(story)
    print(f"✓ PDF report saved to {output_path}")


if __name__ == '__main__':
    generate_report()
